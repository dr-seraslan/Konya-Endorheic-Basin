/**************************************************************
 CORRECTED GEE WORKFLOW
 Native 1 km MODIS LST analytical scale
 No LST downscaling to 30 m
**************************************************************/

/***** 1. DEFINE STUDY AREA / SUB-BASINS *****/

// Replace this asset with your own Konya Endorheic Basin or sub-basin asset.
var subbasins = ee.FeatureCollection('projects/your-project/assets/your-study-area');

// If your sub-basin attribute names are different, edit these two fields.
var SUBBASIN_ID_FIELD = 'subbasin_no';
var SUBBASIN_NAME_FIELD = 'subbasin_name';

var geom = subbasins.geometry();

Map.centerObject(subbasins, 7);
Map.addLayer(subbasins, {color: 'red'}, 'Konya Endorheic Basin / Sub-basins', true);


/***** 2. LOAD AQUA MODIS MYD21A1D LST DATA *****/

var MYD21_raw = ee.ImageCollection('MODIS/061/MYD21A1D')
  .filterBounds(geom)
  .filterDate('2003-01-01', '2024-12-31');

// Use the native MODIS LST projection as the target analytical grid.
// This is the key correction: the study is performed at 1 km, not 30 m.
var targetProj = ee.Image(MYD21_raw.first()).select('LST_1KM').projection();

print('Native MODIS LST projection:', targetProj);


/***** 3. LST PRE-PROCESSING — NO DOWNSCALING *****/

function prepMYD21(img) {
  var qc = img.select('QC');

  // Mandatory QA bits: retain good and nominal quality pixels.
  // This follows the logic of the previous script.
  var mandatoryQA = qc.bitwiseAnd(3);
  var modlandOK = mandatoryQA.lt(2);

  // Cloud-related QA filtering.
  var cloudOK = qc.rightShift(4).bitwiseAnd(3).eq(0);

  // Accuracy-related QA rule retained from the previous version.
  var accOK = qc.rightShift(14).bitwiseAnd(3).gte(2);

  // View angle filtering.
  var vza = img.select('View_Angle');
  var vzaOK = vza.abs().lt(45);

  // Convert MODIS LST to Celsius.
  // IMPORTANT: No .resample('bilinear') is used here.
  // LST remains at its native MODIS 1 km scale.
  var lstC = img.select('LST_1KM')
    .multiply(0.02)
    .subtract(273.15)
    .rename('LST_C')
    .setDefaultProjection(targetProj);

  var mask = modlandOK.and(cloudOK).and(accOK).and(vzaOK);

  return lstC
    .updateMask(mask)
    .copyProperties(img, ['system:time_start']);
}

var myd21C = MYD21_raw.map(prepMYD21);


/***** 4. TOPOGRAPHIC DATA — AGGREGATED TO 1 KM *****/

// Native SRTM is 30 m, but it is aggregated to the MODIS 1 km grid.
// This avoids pseudo-replication and prevents artificial spatial detail.
var SRTM_raw = ee.Image('USGS/SRTMGL1_003')
  .select('elevation')
  .clip(geom);

var elev1km = SRTM_raw
  .reduceResolution({
    reducer: ee.Reducer.mean(),
    maxPixels: 2048
  })
  .reproject({
    crs: targetProj,
    scale: 1000
  })
  .rename('elevation_m')
  .clip(geom);

// Terrain variables are derived from the 1 km harmonized DEM.
var terrain1km = ee.Algorithms.Terrain(elev1km);
var slope1km = terrain1km.select('slope');
var aspect1km = terrain1km.select('aspect');


/***** 5. ELEVATION AND ASPECT CLASSIFICATION AT 1 KM *****/

var elevationBands = [
  {min: 900,  max: 1300, id: 1},
  {min: 1300, max: 1700, id: 2},
  {min: 1700, max: 2100, id: 3},
  {min: 2100, max: 2500, id: 4},
  {min: 2500, max: 2900, id: 5},
  {min: 2900, max: 3400, id: 6}
];

var elevClass = elev1km.multiply(0).toByte().rename('elevation_class');

elevationBands.forEach(function(b, i) {
  var cond = (i === elevationBands.length - 1)
    ? elev1km.gte(b.min).and(elev1km.lte(b.max))
    : elev1km.gte(b.min).and(elev1km.lt(b.max));

  elevClass = elevClass.where(cond, b.id);
});

elevClass = elevClass.updateMask(elevClass.gt(0));

var aspectClass = aspect1km.multiply(0).toByte().rename('aspect_class');

// 5 = Flat
aspectClass = aspectClass.where(slope1km.lt(2), 5);

// 1 = North
aspectClass = aspectClass.where(
  slope1km.gte(2).and(aspect1km.gte(315).or(aspect1km.lt(45))),
  1
);

// 2 = East
aspectClass = aspectClass.where(
  slope1km.gte(2).and(aspect1km.gte(45).and(aspect1km.lt(135))),
  2
);

// 3 = South
aspectClass = aspectClass.where(
  slope1km.gte(2).and(aspect1km.gte(135).and(aspect1km.lt(225))),
  3
);

// 4 = West
aspectClass = aspectClass.where(
  slope1km.gte(2).and(aspect1km.gte(225).and(aspect1km.lt(315))),
  4
);

aspectClass = aspectClass.updateMask(aspectClass.gt(0));


/***** 6. LAND COVER DATA — AGGREGATED FROM 500 M TO 1 KM *****/

var lcCollection = ee.ImageCollection('MODIS/061/MCD12Q1')
  .select('LC_Type1');

function processLC(year) {
  year = ee.Number(year);

  // MCD12Q1 may not be available for the final year.
  // If 2024 is not available, 2023 land cover is used as the closest available year.
  var lcImage = ee.Image(
    ee.Algorithms.If(
      year.gt(2023),
      lcCollection.filter(ee.Filter.calendarRange(2023, 2023, 'year')).first(),
      lcCollection.filter(ee.Filter.calendarRange(year, year, 'year')).first()
    )
  );

  // Aggregate 500 m categorical land cover to 1 km using mode.
  // This preserves categorical integrity better than bilinear interpolation.
  var lc1km = lcImage
    .reduceResolution({
      reducer: ee.Reducer.mode(),
      maxPixels: 16
    })
    .reproject({
      crs: targetProj,
      scale: 1000
    })
    .rename('LC_Type1_1km')
    .clip(geom);

  var lcClass = lc1km.multiply(0).toByte().rename('landcover_class');

  // Reclassification based on IGBP LC_Type1.
  lcClass = lcClass.where(lc1km.eq(12).or(lc1km.eq(14)), 1); // Cropland
  lcClass = lcClass.where(lc1km.eq(16), 2);                  // Barren
  lcClass = lcClass.where(lc1km.eq(10), 3);                  // Grassland
  lcClass = lcClass.where(
    lc1km.eq(6).or(lc1km.eq(7)).or(lc1km.eq(8)).or(lc1km.eq(9)),
    4
  );                                                        // Shrubland / Savanna
  lcClass = lcClass.where(lc1km.gte(1).and(lc1km.lte(5)), 5); // Forest
  lcClass = lcClass.where(lc1km.eq(13), 6);                  // Urban

  // Water and unclassified pixels remain masked.
  return lcClass.updateMask(lcClass.gt(0)).clip(geom);
}


/***** 7. LABEL DICTIONARIES *****/

var elevLabels = ee.Dictionary({
  '1': '900-1300m',
  '2': '1300-1700m',
  '3': '1700-2100m',
  '4': '2100-2500m',
  '5': '2500-2900m',
  '6': '2900-3400m'
});

var aspectLabels = ee.Dictionary({
  '1': 'North',
  '2': 'East',
  '3': 'South',
  '4': 'West',
  '5': 'Flat'
});

var lcLabels = ee.Dictionary({
  '1': 'Cropland',
  '2': 'Barren_Sparse_Vegetation',
  '3': 'Grassland',
  '4': 'Shrubland_Savanna',
  '5': 'Forest',
  '6': 'Urban'
});


/***** 8. ANNUAL MEDIAN DAYTIME LST EXTRACTION AT 1 KM *****/

var yearsToProcess = ee.List.sequence(2003, 2024);
var nSubbasins = subbasins.size();
var subbasinList = subbasins.toList(nSubbasins);

function extractForYearAndSubbasin(runYear, subbasinFeature) {
  runYear = ee.Number(runYear);
  subbasinFeature = ee.Feature(subbasinFeature);

  var annualCollection = myd21C
    .filter(ee.Filter.calendarRange(runYear, runYear, 'year'));

  // Annual median daytime LST composite.
  // No seasonal filtering is applied.
  var annualMedianLST = annualCollection
    .median()
    .rename('LST_C')
    .setDefaultProjection(targetProj);

  var lcYear = processLC(runYear);

  // Combined bio-physical zone ID:
  // elevation_class * 100 + landcover_class * 10 + aspect_class
  var combo = elevClass
    .multiply(100)
    .add(lcYear.multiply(10))
    .add(aspectClass)
    .rename('combo_id')
    .toInt16()
    .setDefaultProjection(targetProj);

  var reducer = ee.Reducer.median()
    .combine({
      reducer2: ee.Reducer.count(),
      sharedInputs: true
    });

  // IMPORTANT:
  // scale is 1000 m, not 30 m.
  var grouped = ee.Algorithms.If(
    annualCollection.size().gt(0),
    annualMedianLST.addBands(combo).reduceRegion({
      reducer: reducer.group({
        groupField: 1,
        groupName: 'combo_id'
      }),
      geometry: subbasinFeature.geometry(),
      scale: 1000,
      crs: targetProj,
      maxPixels: 1e13,
      tileScale: 8
    }).get('groups'),
    ee.List([])
  );

  grouped = ee.List(grouped);

  var features = grouped.map(function(d) {
    d = ee.Dictionary(d);

    var comboID = ee.Number(d.get('combo_id')).toInt();

    var elevID = comboID.divide(100).floor();
    var lcID = comboID.mod(100).divide(10).floor();
    var aspectID = comboID.mod(10);

    return ee.Feature(null).set({
      'year': runYear,

      'subbasin_no': subbasinFeature.get(SUBBASIN_ID_FIELD),
      'subbasin_name': subbasinFeature.get(SUBBASIN_NAME_FIELD),

      'combo_id': comboID,

      'elevation_class': elevID,
      'elevation_class_name': elevLabels.get(elevID.format()),

      'landcover_class': lcID,
      'landcover_name': lcLabels.get(lcID.format()),

      'aspect_class': aspectID,
      'aspect_class_name': aspectLabels.get(aspectID.format()),

      'LST_median_C': d.get('median'),
      'pixel_count_1km': d.get('count')
    });
  });

  return ee.FeatureCollection(features);
}


function extractForYear(runYear) {
  var perSubbasinCollections = ee.List.sequence(0, nSubbasins.subtract(1))
    .map(function(i) {
      var sb = ee.Feature(subbasinList.get(i));
      return extractForYearAndSubbasin(runYear, sb);
    });

  return ee.FeatureCollection(perSubbasinCollections).flatten();
}

var labeledFeatures = ee.FeatureCollection(
  yearsToProcess.map(function(y) {
    return extractForYear(y);
  })
).flatten();

print('Annual 1 km LST zonal statistics:', labeledFeatures.limit(10));
